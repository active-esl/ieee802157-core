cmake_minimum_required(VERSION 3.20)
project(ieee802157_core VERSION 0.1.0
  DESCRIPTION "Portable experimental optical camera communications core"
  HOMEPAGE_URL "https://github.com/active-esl/ieee802157-core" LANGUAGES C)
option(OCC_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(CTest)
add_library(occ STATIC src/occ.c src/packet.c src/receiver.c src/image.c src/player.c src/spatial.c)
target_sources(occ PRIVATE src/cook.c src/rolling.c)
target_include_directories(occ PUBLIC include)
target_compile_features(occ PUBLIC c_std_99)
set_target_properties(occ PROPERTIES C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(occ PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(OCC_ENABLE_SANITIZERS)
    target_compile_options(occ PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(occ PUBLIC -fsanitize=address,undefined)
  endif()
elseif(OCC_ENABLE_SANITIZERS)
  message(FATAL_ERROR "OCC_ENABLE_SANITIZERS requires GCC or Clang")
endif()
if(BUILD_TESTING)
  add_executable(test_occ tests/test_occ.c)
  target_link_libraries(test_occ PRIVATE occ)
  add_test(NAME core COMMAND test_occ)
  add_executable(test_receiver tests/test_receiver.c)
  target_link_libraries(test_receiver PRIVATE occ)
  add_test(NAME receiver COMMAND test_receiver)
  add_executable(test_image tests/test_image.c)
  target_link_libraries(test_image PRIVATE occ)
  add_test(NAME image COMMAND test_image)
  add_executable(test_player tests/test_player.c)
  target_link_libraries(test_player PRIVATE occ)
  add_test(NAME player COMMAND test_player)
  add_executable(test_spatial tests/test_spatial.c)
  target_link_libraries(test_spatial PRIVATE occ)
  add_test(NAME spatial COMMAND test_spatial)
  add_executable(test_cook tests/test_cook.c)
  target_link_libraries(test_cook PRIVATE occ)
  add_test(NAME cook COMMAND test_cook)
  add_executable(test_hardening tests/test_hardening.c)
  target_link_libraries(test_hardening PRIVATE occ)
  add_test(NAME hardening COMMAND test_hardening)
  set_tests_properties(core receiver image player spatial cook hardening PROPERTIES TIMEOUT 30)
endif()
add_executable(occ_wave_csv examples/linux/wave_csv.c)
target_link_libraries(occ_wave_csv PRIVATE occ)
add_executable(occ_fixture_json examples/linux/fixture_json.c)
target_link_libraries(occ_fixture_json PRIVATE occ)
add_executable(occ_decode_gray examples/linux/decode_gray.c)
target_link_libraries(occ_decode_gray PRIVATE occ)
add_executable(occ_decode_spatial examples/linux/decode_spatial.c)
target_link_libraries(occ_decode_spatial PRIVATE occ)
add_executable(occ_fixture_cook examples/linux/fixture_cook.c)
target_link_libraries(occ_fixture_cook PRIVATE occ)
add_executable(occ_decode_rows examples/linux/decode_rows.c)
target_link_libraries(occ_decode_rows PRIVATE occ)
add_executable(occ_measure examples/linux/measure.c)
target_link_libraries(occ_measure PRIVATE occ)
add_executable(occ_status_stream examples/linux/status_stream.c)
target_link_libraries(occ_status_stream PRIVATE occ)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(occ_status_stream PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME status_stream COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_status_stream.py $<TARGET_FILE:occ_status_stream>)
    set_tests_properties(status_stream PROPERTIES TIMEOUT 30
      ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  endif()
endif()
